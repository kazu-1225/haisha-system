📌 Googleスプレッドシート連携とは？
データをGoogle スプレッドシートに保存することで、社内パソコン・タブレット・スマホ・外出先などどこからでも同じデータにアクセスできます。Googleアカウントがあれば無料で利用できます。
🔗 現在の接続状態
⚪ URLが設定されていません
📋 初回セットアップ手順（一度だけ行ってください）
1
Googleスプレッドシートを新規作成
sheets.google.com を開き、「＋ 新しいスプレッドシート」を作成してください。
ファイル名は「配車管理DB」などわかりやすい名前にしてください。
2
Apps Script を開く
スプレッドシートのメニューから「拡張機能 → Apps Script」を選択してください。
3
下のコードをコピーして貼り付け
エディタの既存コードを全て削除し、下のコードを貼り付けて「💾 保存」してください。
// ===================================================== // 配車管理システム - Google Apps Script バックエンド // このコードをApps Scriptエディタに貼り付けてください // ===================================================== const SHEET_NAME = 'DB_JSON'; function doGet(e) { try { const data = readData(); return respond(data); } catch(err) { return respond({ error: err.message }); } } function doPost(e) { try { const data = JSON.parse(e.postData.contents); writeData(data); writeReadableSheets(data); return respond({ ok: true }); } catch(err) { return respond({ ok: false, error: err.message }); } } function respond(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function readData() { const sheet = getSheet(SHEET_NAME); const val = sheet.getRange('A1').getValue(); if (!val) return { drivers: [], vehicles: [], customers: [], regulars: [], dispatches: [] }; return JSON.parse(val); } function writeData(data) { getSheet(SHEET_NAME).getRange('A1').setValue(JSON.stringify(data)); } function getSheet(name) { const ss = SpreadsheetApp.getActiveSpreadsheet(); return ss.getSheetByName(name) || ss.insertSheet(name); } function writeReadableSheets(data) { const DOW = ['日','月','火','水','木','金','土']; function cust(id) { const c = (data.customers||[]).find(x => x.id === id); return c ? c.name : ''; } function drv(id) { if (!id) return '未定'; const d = (data.drivers||[]).find(x => x.id === id); return d ? d.name : '未定'; } function veh(id) { if (!id) return '未定'; const v = (data.vehicles||[]).find(x => x.id === id); return v ? v.name : '未定'; } // 配車一覧シート const dispRows = (data.dispatches||[]).map(dp => { const d = dp.date ? new Date(dp.date) : null; return [ dp.date || '', d ? DOW[d.getDay()] : '', cust(dp.customer), drv(dp.driver), veh(dp.vehicle), dp.isReg ? '定期' : 'スポット', dp.status === 'confirmed' ? '確定' : '保留', dp.notes || '', (dp.created||'').slice(0,10) ]; }); writeSheet('📋配車一覧', ['日付','曜日','お客様','ドライバー','車両','種別','ステータス','備考','登録日'], dispRows ); // ドライバー一覧 writeSheet('👤ドライバー', ['ID','名前','保有免許'], (data.drivers||[]).map(d => [d.id, d.name, (d.license||[]).join('・')]) ); // 車両一覧 writeSheet('🚛車両', ['ID','名前','ナンバー','種別','必要免許'], (data.vehicles||[]).map(v => [v.id, v.name, v.plate||'', v.type||'', v.req||'']) ); // お客様一覧 writeSheet('🏢お客様', ['ID','お客様名','住所','連絡先'], (data.customers||[]).map(c => [c.id, c.name, c.addr||'', c.contact||'']) ); } function writeSheet(name, headers, rows) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(name); if (!sheet) sheet = ss.insertSheet(name); sheet.clearContents(); const hRow = sheet.getRange(1, 1, 1, headers.length); hRow.setValues([headers]); hRow.setFontWeight('bold'); hRow.setBackground('#1e3a5f'); hRow.setFontColor('white'); if (rows.length > 0) { sheet.getRange(2, 1, rows.length, headers.length).setValues(rows); } try { sheet.autoResizeColumns(1, headers.length); } catch(e) {} }
4
Webアプリとしてデプロイ
Apps Script 画面の右上「デプロイ → 新しいデプロイ」をクリック
①「種類の選択」→「ウェブアプリ」
②「次のユーザーとして実行」→「自分」
③「アクセスできるユーザー」→「全員」
④「デプロイ」ボタンを押して、表示された URL をコピー
5
URLを上の入力欄に貼り付けて保存
コピーしたURLを、このページ上部の「Google Apps Script の Web アプリURL」欄に貼り付けて「✅ URLを保存・接続テスト」をクリックしてください。
6
他の端末でも同じHTMLファイルを開いてURLを設定
社内の他のパソコン・タブレットでも、この 配車管理システム.html を開き、同じURLを設定すれば完了です。
