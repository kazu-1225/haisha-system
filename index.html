📌 Googleスプレッドシート連携とは？
データをGoogle スプレッドシートに保存することで、社内パソコン・タブレット・スマホ・外出先などどこからでも同じデータにアクセスできます。Googleアカウントがあれば無料で利用できます。
🔗 現在の接続状態
⚪ URLが設定されていません
📋 初回セットアップ手順（一度だけ行ってください）
1
Googleスプレッドシートを新規作成
sheets.google.com を開き、「＋ 新しいスプレッドシート」を作成してください。
ファイル名は「配車管理DB」などわかりやすい名前にしてください。
2
Apps Script を開く
スプレッドシートのメニューから「拡張機能 → Apps Script」を選択してください。
3
下のコードをコピーして貼り付け
エディタの既存コードを全て削除し、下のコードを貼り付けて「💾 保存」してください。
// ===================================================== // 配車管理システム - Google Apps Script バックエンド // このコードをApps Scriptエディタに貼り付けてください // ===================================================== const SHEET_NAME = 'DB_JSON'; function doGet(e) { try { const data = readData(); return respond(data); } catch(err) { return respond({ error: err.message }); } } function doPost(e) { try { const data = JSON.parse(e.postData.contents); writeData(data); writeReadableSheets(data); return respond({ ok: true }); } catch(err) { return respond({ ok: false, error: err.message }); } } function respond(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function readData() { const sheet = getSheet(SHEET_NAME); const val = sheet.getRange('A1').getValue(); if (!val) return { drivers: [], vehicles: [], customers: [], regulars: [], dispatches: [] }; return JSON.parse(val); } function writeData(data) { getSheet(SHEET_NAME).getRange('A1').setValue(JSON.stringify(data)); } function getSheet(name) { const ss = SpreadsheetApp.getActiveSpreadsheet(); return ss.getSheetByName(name) || ss.insertSheet(name); } function writeReadableSheets(data) { const DOW = ['日','月','火','水','木','金','土']; function custName(id) { const c = (data.customers||[]).find(x => x.id === id); return c ? c.name : (id||''); } function drvName(id) { if (!id) return '未定'; const d = (data.drivers||[]).find(x => x.id === id); return d ? d.name : '未定'; } function vehName(id) { if (!id) return '未定'; const v = (data.vehicles||[]).find(x => x.id === id); return v ? v.name : '未定'; } function fmtDate(s) { if (!s) return ''; const d = new Date(s); return isNaN(d.getTime()) ? s : (d.getFullYear()+'/'+String(d.getMonth()+1).padStart(2,'0')+'/'+String(d.getDate()).padStart(2,'0')); } function dow(s) { if (!s) return ''; const d = new Date(s); return isNaN(d.getTime()) ? '' : DOW[d.getDay()]; } // ===== 📋 配車一覧（全件） ===== const dispAll = (data.dispatches||[]).map(dp => { const wDate = dp.dispatchDate || dp.date || ''; const wt = dp.workType || {}; const wtStr = wt.type === 'container' ? 'コンテナ(' + (wt.containerOp||'') + '/' + (wt.containerSize||'') + ')' : wt.type === 'manual' ? '手積み' : ''; const mf = dp.manifest || {}; return [ fmtDate(wDate), dow(wDate), custName(dp.customer), dp.custContact || '', dp.custPhone || '', dp.siteName || '', dp.siteContact || '', dp.sitePhone || '', dp.siteAddr || '', drvName(dp.driver), vehName(dp.vehicle), dp.isCarryIn ? '持込み' : dp.isReg ? '定期便' : 'スポット', dp.status === 'confirmed' ? '確定' : '保留', wtStr, mf.needed === 'yes' || mf.needed === 'bring' || mf.needed === 'receive' ? '要' : '不要', dp.notes || '', dp.caution || '', dp.receptionist || '', fmtDate(dp.date), (dp.deadline || ''), (dp.created||'').slice(0,10) ]; }); writeSheet('📋配車一覧', ['配車確定日','曜日','お客様名','お客様担当者','担当者電話','収集先名称','収集先担当者','収集先電話','収集先住所', 'ドライバー','車両','種別','ステータス','作業種別','マニフェスト','備考','注意事項','受付担当者','受付日','期限','登録日'], dispAll ); // ===== ✅ 配車実績（確定済みのみ） ===== const dispDone = (data.dispatches||[]).filter(dp => dp.status === 'confirmed' && (dp.dispatchDate || dp.date) ).map(dp => { const wDate = dp.dispatchDate || dp.date || ''; return [ fmtDate(wDate), dow(wDate), custName(dp.customer), dp.custContact || '', dp.custPhone || '', dp.siteName || '', dp.siteContact || '', dp.sitePhone || '', drvName(dp.driver), vehName(dp.vehicle), dp.isCarryIn ? '持込み' : dp.isReg ? '定期便' : 'スポット', dp.notes || '', dp.receptionist || '', fmtDate(dp.date) ]; }); writeSheet('✅配車実績', ['配車日','曜日','お客様名','お客様担当者','担当者電話','収集先名称','収集先担当者','収集先電話', 'ドライバー','車両','種別','備考','受付担当者','受付日'], dispDone ); // ===== 🏢 お客様マスタ ===== writeSheet('🏢お客様', ['ID','T-PlannerCD','お客様名','住所','担当者名','電話番号'], (data.customers||[]).map(c => [c.id, c.tpcd||'', c.name, c.addr||'', c.contact||'', c.phone||'']) ); // ===== 📍 収集先マスタ ===== writeSheet('📍収集先', ['ID','お客様ID','収集先名称','担当者名','電話番号','住所','注意事項'], (data.sites||[]).map(s => [s.id, s.customerId||'', s.name, s.contact||'', s.phone||'', s.addr||'', s.caution||'']) ); // ===== 👤 ドライバーマスタ ===== writeSheet('👤ドライバー', ['ID','名前','グループ','種別','保有免許'], (data.drivers||[]).map(d => [d.id, d.name, [d.group,d.group2].filter(x=>x).join('/'), d.type||'', (d.license||[]).join('・')]) ); // ===== 🚛 車両マスタ ===== writeSheet('🚛車両', ['ID','車両名','ナンバー','車種','必要免許'], (data.vehicles||[]).map(v => [v.id, v.name, v.plate||'', v.type||'', v.req||'']) ); } function writeSheet(name, headers, rows) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(name); if (!sheet) sheet = ss.insertSheet(name); sheet.clearContents(); const hRow = sheet.getRange(1, 1, 1, headers.length); hRow.setValues([headers]); hRow.setFontWeight('bold'); hRow.setBackground('#1e3a5f'); hRow.setFontColor('white'); if (rows.length > 0) { sheet.getRange(2, 1, rows.length, headers.length).setValues(rows); } try { sheet.autoResizeColumns(1, headers.length); } catch(e) {} }
4
Webアプリとしてデプロイ
Apps Script 画面の右上「デプロイ → 新しいデプロイ」をクリック
①「種類の選択」→「ウェブアプリ」
②「次のユーザーとして実行」→「自分」
③「アクセスできるユーザー」→「全員」
④「デプロイ」ボタンを押して、表示された URL をコピー
5
URLを上の入力欄に貼り付けて保存
コピーしたURLを、このページ上部の「Google Apps Script の Web アプリURL」欄に貼り付けて「✅ URLを保存・接続テスト」をクリックしてください。
6
他の端末でも同じHTMLファイルを開いてURLを設定
社内の他のパソコン・タブレットでも、この 配車管理システム.html を開き、同じURLを設定すれば完了です。
💡 ここで登録した収集先は、配車編集画面の「収集先情報」ドロップダウンからすぐに呼び出せます。工事現場など頻繁に変わる現場もここで管理してください。
